\section{Visão Geral}
%==============================================================================

O modelo de otimização tem como objetivo maximizar a margem total de contribuição na alocação da produção semanal de ovos para diferentes SKUs (\textit{Stock Keeping Units}), respeitando:

\begin{itemize}
    \item Capacidade de produção por classe de produto
    \item Demanda histórica como limite superior por SKU
    \item Reservas prioritárias para pedidos garantidos
\end{itemize}

O resultado é comparado com um \textbf{baseline} que representa a margem obtida caso a empresa mantivesse a distribuição histórica de vendas.

%==============================================================================
\section{Conjuntos e Índices}
%==============================================================================

\subsection{Conjuntos Principais}

\begin{align}\label{eq:auto:al:001}
I &= \{1, 2, \ldots, n\} && \text{Conjunto de todos os SKUs} \\
C &= \{c_1, c_2, \ldots, c_m\} && \text{Conjunto de classes de produtos} \\
S &= \{s_1, s_2, \ldots, s_k\} && \text{Conjunto de períodos (semanas) no histórico}
\end{align}

\subsection{Subconjuntos Derivados}

\begin{align}\label{eq:auto:al:002}
I_c &\subseteq I && \text{SKUs pertencentes à classe } c \\
I_{\text{válido}} &\subseteq I && \text{SKUs válidos para otimização} \\
H &\subseteq I_{\text{válido}} && \text{SKUs com demanda histórica válida} \\
R &\subseteq I && \text{SKUs com pedidos reservados}
\end{align}

\subsection{Mapeamentos}

\begin{align}\label{eq:auto:al:003}
\kappa: I &\rightarrow C && \text{Função que mapeia SKU para sua classe} \\
e_i: I &\rightarrow \mathbb{Z}^+ && \text{Quantidade de ovos por embalagem do SKU } i
\end{align}

\section{Parâmetros de Entrada}
%==============================================================================

\subsection{Dados do SKU}

\begin{table}[h]
\centering
\begin{tabular}{clll}
\toprule
\textbf{Símbolo} & \textbf{Descrição} & \textbf{Unidade} & \textbf{Domínio} \\
\midrule
$p_i$ & Preço de venda do SKU $i$ & R\$/caixa & $\mathbb{R}^+$ \\
$c_i$ & Custo YTD do SKU $i$ & R\$/caixa & $\mathbb{R}^+$ \\
$e_i$ & Quantidade de ovos por caixa & ovos/caixa & $\mathbb{Z}^+$ \\
$\kappa(i)$ & Classe do SKU $i$ & - & $C$ \\
\bottomrule
\end{tabular}
\caption{Parâmetros por SKU}
\end{table}

\subsection{Cálculo da Embalagem}

A quantidade de ovos por embalagem $e_i$ é extraída da descrição do produto:

\begin{equation}\label{eq:auto:001}
e_i = n_{\text{bandejas}} \times n_{\text{unidades}}
\end{equation}

Exemplo: ``CX 12 BJ 30 UN'' $\Rightarrow e_i = 12 \times 30 = 360$ ovos.

\subsection{Dados de Produção}

\begin{equation}\label{eq:auto:002}
P_c \in \mathbb{R}^+ \quad \text{Produção total da classe } c \text{ na semana (ovos)}
\end{equation}

\subsection{Margem de Contribuição}

\textbf{Margem por caixa:}
\begin{equation}\label{eq:auto:003}
m_i = p_i - c_i \quad \text{(R\$/caixa)}
\end{equation}

\textbf{Margem por ovo} (coeficiente da função objetivo):
\begin{equation}\label{eq:auto:004}
\mu_i = \frac{m_i}{e_i} = \frac{p_i - c_i}{e_i} \quad \text{(R\$/ovo)}
\end{equation}

\textbf{Tratamento de custo ausente:}

Quando um SKU $i$ não possui custo individual, utiliza-se a média da classe:
\begin{equation}\label{eq:auto:005}
c_i = \begin{cases}
c_i^{\text{original}} & \text{se } c_i^{\text{original}} > 0 \\[4pt]
\displaystyle\frac{\sum_{j \in I_{\kappa(i)}} c_j}{|I_{\kappa(i)}|} & \text{caso contrário}
\end{cases}
\end{equation}

\section{Processamento de Dados (ETL)}
%==============================================================================

\subsection{Base de Faturamento}

Cada registro de faturamento contém item, cliente, estabelecimento, data e quantidade em caixas. A quantidade é convertida para ovos:

\begin{equation}\label{eq:auto:006}
Q_f^{\text{ovos}} = Q_f^{\text{caixas}} \times 360
\end{equation}

\subsection{Correção de Estabelecimento}

Alguns registros têm estabelecimento incorreto. A correção é feita via mapeamento externo:

\begin{equation}\label{eq:auto:007}
\text{Estab\_Corrigido}(f) = \begin{cases}
E(\text{cliente}_f) & \text{se } \text{cliente}_f \in \text{dom}(E) \\
\text{estab}_f & \text{caso contrário}
\end{cases}
\end{equation}

onde $E: \text{Cod.Cliente} \rightarrow \text{Estab\_Padrao}$ é o mapeamento de correção.

\subsection{Filtro por Estabelecimento}

Apenas registros do estabelecimento de interesse:
\begin{equation}\label{eq:auto:008}
F_{\text{filtrado}} = \{f \in F : \text{Estab\_Corrigido}(f) = 100\}
\end{equation}

\subsection{Janela Temporal}

Os dados são filtrados por uma janela configurável:
\begin{equation}\label{eq:auto:009}
\text{Janela} = [\text{mes\_ref} - \text{meses\_janela} + 1, \; \text{mes\_ref}]
\end{equation}

Configuração atual: \texttt{mes\_custo=11}, \texttt{meses\_janela\_custo=6} $\Rightarrow$ junho a novembro de 2025.

\subsection{Agregação por Período}

A granularidade é configurável (\texttt{S}=semanal, \texttt{M}=mensal, \texttt{D}=diário). Para cada SKU $i$ e período $s$:

\begin{equation}\label{eq:auto:010}
Q_{i,s} = \sum_{f : \text{item}_f = i, \, \text{período}(f) = s} Q_f^{\text{ovos}}
\end{equation}

\subsection{Cálculo da Demanda Histórica}

O limite de demanda depende do tipo de cálculo configurado:

\textbf{Método Máximo} (\texttt{tipo\_calculo\_demanda = maximo}):
\begin{equation}\label{eq:auto:011}
D_i^{\text{base}} = \max_{s \in S_i} Q_{i,s}
\end{equation}

\textbf{Método Média} (\texttt{tipo\_calculo\_demanda = media}):
\begin{equation}\label{eq:auto:012}
D_i^{\text{base}} = \frac{1}{|S_i|} \sum_{s \in S_i} Q_{i,s}
\end{equation}

\textbf{Método Percentil} (\texttt{tipo\_calculo\_demanda = percentil}):
\begin{equation}\label{eq:auto:013}
D_i^{\text{base}} = \text{Percentil}_p(Q_{i,s}) \quad \text{onde } p = \texttt{percentil\_demanda}
\end{equation}

onde $S_i = \{s \in S : Q_{i,s} > 0\}$ é o conjunto de períodos com vendas.

\textbf{Aplicação do fator de expansão:}
\begin{equation}\label{eq:auto:014}
D_i = D_i^{\text{base}} \times \alpha \quad \text{onde } \alpha = \texttt{fator\_demanda\_maxima}
\end{equation}

O fator $\alpha$ é aplicado igualmente a todos os SKUs, independente do método.

\subsection{Volume Histórico Total}

Para o cálculo do baseline, armazena-se o volume total vendido por SKU no período (sem fator):

\begin{equation}\label{eq:auto:015}
V_i^{\text{hist}} = \sum_{s \in S_i} Q_{i,s}
\end{equation}

\subsection{Classificação de SKUs Válidos}

Um SKU é \textbf{válido para otimização} se:

\begin{equation}\label{eq:auto:016}
i \in I_{\text{válido}} \iff \begin{cases}
\text{ESTAB}_i = 100 \\
\text{STATUS}_i = \texttt{ATIVO} \\
\text{TIPO}_i \notin \{\texttt{GRANEL}, \texttt{Exportação}, \texttt{MARCA PROPRIA}\} \\
\text{CLASSE}_i \neq \texttt{OUTROS} \\
e_i > 0
\end{cases}
\end{equation}

SKUs da classe \texttt{OUTROS} são excluídos da otimização mas aparecem no output com alocação zero.

\section{Sistema de Reservas}
%==============================================================================

\subsection{Critério de Elegibilidade}

Um pedido para SKU $i$ é reservado se o SKU tem classe mapeada e produção na semana:

\begin{equation}\label{eq:auto:017}
i \in R \iff (Q_i^{\text{pedido}} > 0) \land (\kappa(i) \text{ existe}) \land (P_{\kappa(i)} > 0)
\end{equation}

\subsection{Quantidade Reservada}

\begin{equation}\label{eq:auto:018}
r_i = Q_i^{\text{pedido}} \quad \forall i \in R
\end{equation}

\subsection{Reservas por Classe}

\begin{equation}\label{eq:auto:019}
R_c = \sum_{i \in R : \kappa(i) = c} r_i \quad \forall c \in C
\end{equation}

\subsection{Produção Excedente}

Volume disponível para otimização após descontar reservas:

\begin{equation}\label{eq:auto:020}
E_c = \max(0, \; P_c - R_c) \quad \forall c \in C
\end{equation}

%==============================================================================
\section{Modelo de Otimização}
%==============================================================================

\subsection{Variável de Decisão}

\begin{equation}\label{eq:auto:021}
x_i \geq 0 \quad \text{Quantidade alocada ao SKU } i \text{ (em ovos)}
\end{equation}

\textbf{Tipo da variável} (configurável via \texttt{variaveis\_continuas}):
\begin{equation}\label{eq:auto:022}
x_i \in \begin{cases}
\mathbb{R}^+ & \text{se \texttt{variaveis\_continuas = true} (padrão)} \\
\mathbb{Z}^+ & \text{se \texttt{variaveis\_continuas = false}}
\end{cases}
\end{equation}

\subsection{Função Objetivo}

Maximizar a margem total de contribuição:

\begin{equation}\label{eq:auto:023}
\max \; Z = \sum_{i \in H} \mu_i \cdot x_i = \sum_{i \in H} \frac{p_i - c_i}{e_i} \cdot x_i
\end{equation}

\subsection{Restrições}

\subsubsection{R1: Capacidade de Produção por Classe}

\begin{equation}\label{eq:auto:024}
\sum_{i \in H : \kappa(i) = c} x_i \leq E_c \quad \forall c \in C
\end{equation}

\subsubsection{R2: Limite de Demanda Histórica}

Ativa quando \texttt{considerar\_demanda\_historica = true}:

\begin{equation}\label{eq:auto:025}
x_i \leq D_i \quad \forall i \in H
\end{equation}

\subsubsection{R3: Não-Negatividade}

\begin{equation}\label{eq:auto:026}
x_i \geq 0 \quad \forall i \in H
\end{equation}

\subsection{Formulação Compacta}

\begin{equation}\label{eq:auto:027}
\boxed{
\begin{aligned}
\max \quad & Z = \sum_{i \in H} \mu_i \cdot x_i \\[10pt]
\text{sujeito a:} \quad & \\[4pt]
& \sum_{i \in H : \kappa(i) = c} x_i \leq E_c && \forall c \in C \quad \text{(capacidade)} \\[6pt]
& x_i \leq D_i && \forall i \in H \quad \text{(demanda histórica)} \\[6pt]
& x_i \geq 0 && \forall i \in H \quad \text{(não-negatividade)}
\end{aligned}
}
\end{equation}

\section{Comparativo: Baseline vs Otimizado}
%==============================================================================

O comparativo avalia o ganho da otimização em relação a um cenário \textbf{baseline} que representa a distribuição histórica real das vendas.

\subsection{Princípio}

\begin{itemize}
    \item \textbf{Baseline}: ``Qual seria a margem se continuássemos vendendo na mesma proporção histórica?''
    \item \textbf{Otimizado}: ``Qual a margem obtida pelo modelo ao realocar volume para SKUs mais rentáveis?''
\end{itemize}

Ambos utilizam o \textbf{mesmo volume total} por classe, excluindo reservas e classe OUTROS.

\subsection{Volume Utilizado}

O volume comparável por classe é o total alocado pelo modelo (excluindo reservas e OUTROS):

\begin{equation}\label{eq:auto:028}
V_c^{\text{otim}} = \sum_{\substack{i \in H : \kappa(i) = c \\ \text{tipo}_i \neq \text{reserva}}} x_i^*
\end{equation}

\subsection{Margem Otimizada}

Resultado direto do solver:

\begin{equation}\label{eq:auto:029}
M^{\text{otim}} = \sum_{\substack{i \in H \\ \text{tipo}_i = \text{otimizacao}}} \mu_i \cdot x_i^*
= \sum_{\substack{i \in H \\ \text{tipo}_i = \text{otimizacao}}} \frac{m_i}{e_i} \cdot x_i^*
\end{equation}

\subsection{Margem Baseline}

\subsubsection{Passo 1: Proporção Histórica}

Para cada classe $c$, calcula-se a proporção de cada SKU com base no volume total vendido no período (sem fator multiplicativo):

\begin{equation}\label{eq:auto:030}
\pi_i = \frac{V_i^{\text{hist}}}{\displaystyle\sum_{j \in H : \kappa(j) = c} V_j^{\text{hist}}}
\quad \forall i \in H : \kappa(i) = c
\end{equation}

onde $V_i^{\text{hist}} = \sum_{s \in S_i} Q_{i,s}$ é o volume total histórico do SKU $i$ na mesma janela temporal usada para o cálculo do limite de demanda.

\textbf{Propriedade}: $\sum_{i \in H : \kappa(i) = c} \pi_i = 1$ para cada classe $c$.

\textbf{Fallback}: Se $\sum_j V_j^{\text{hist}} = 0$ (sem dados históricos), usa-se distribuição uniforme: $\pi_i = 1/|I_c \cap H|$.

\subsubsection{Passo 2: Volume Baseline por SKU}

Distribui o volume disponível da classe proporcionalmente ao histórico:

\begin{equation}\label{eq:auto:031}
x_i^{\text{base}} = V_c^{\text{otim}} \cdot \pi_i \quad \forall i \in H : \kappa(i) = c
\end{equation}

\subsubsection{Passo 3: Margem Baseline Total}

\begin{equation}\label{eq:auto:032}
M^{\text{base}} = \sum_{c \in C} \sum_{i \in H : \kappa(i) = c} x_i^{\text{base}} \cdot \mu_i
= \sum_{c \in C} \sum_{i \in H : \kappa(i) = c} V_c^{\text{otim}} \cdot \pi_i \cdot \frac{m_i}{e_i}
\end{equation}

\subsection{Cálculo do Ganho}

\begin{align}\label{eq:auto:al:004}
\text{Ganho Absoluto} &= M^{\text{otim}} - M^{\text{base}} \\[6pt]
\text{Ganho Percentual} &= \frac{M^{\text{otim}} - M^{\text{base}}}{M^{\text{base}}} \times 100\%
\end{align}

\subsection{Interpretação}

\begin{itemize}
    \item \textbf{Ganho $> 0$}: A otimização gera mais margem que a distribuição histórica, realocando volume para SKUs mais rentáveis.
    \item \textbf{Ganho $= 0$}: A distribuição histórica já é ótima dadas as restrições.
    \item \textbf{Ganho $< 0$}: As restrições (ex: demanda histórica muito restritiva) forçam o modelo a uma distribuição pior que a histórica.
\end{itemize}

\section{Formulação da Distribuição Histórica por Dia da Semana (DOW)}
%==============================================================================

Esta seção formaliza o pós-processamento para granularidade semanal (\texttt{S}),
que distribui os volumes semanais decididos pelo modelo de otimização em estimativas
diárias por SKU, mantendo consistência exata com o output do comparador de
produção/alocação.

\subsection{Objetivo}

Dado o volume semanal decidido pelo modelo para cada SKU, $V^{\text{mod}}_{i,w^*}$,
estimar o volume diário:
\begin{equation}\label{eq:dow:objetivo}
\hat{V}_{i,w^*,d}, \quad \forall i \in \mathcal{I},\;\forall d \in \mathcal{D}
\end{equation}

\textbf{Requisito de consistência:} a soma dos 7 dias deve coincidir exatamente com o
valor exibido no comparador, garantindo que o DOW seja uma \emph{desagregação temporal
fiel} da decisão do modelo.

\subsection{Índices e Conjuntos}

\begin{align}\label{eq:dow:indices}
i &\in \mathcal{I} && \text{SKUs com volume decidido pelo modelo ($V^{\text{mod}}_{i,w^*}>0$)} \nonumber\\
c &\in \mathcal{C} && \text{Classes de produto} \nonumber\\
d &\in \mathcal{D}=\{1,\ldots,7\} && \text{Dia da semana (1=dom,\ldots,7=sáb)} \nonumber\\
w &\in \mathcal{W} && \text{Semanas da janela histórica} \nonumber\\
w^* &&& \text{Semana alvo (configurada em \texttt{semana\_ref})}
\end{align}

Mapeamentos:
\begin{align}\label{eq:dow:mapeamentos}
\kappa(i) &= c && \text{Classe do SKU } i \nonumber\\
\mathcal{I}_c &= \{i \in \mathcal{I} : \kappa(i)=c\} && \text{SKUs da classe } c \text{ com volume no modelo}
\end{align}

\textbf{Premissa:} $\mathcal{I}$ contém apenas SKUs ativos, filtrados pelos mesmos
critérios de estabelecimento e status do pipeline principal.

\subsection{Dados de Entrada}

\subsubsection{Volume decidido pelo modelo (semanal por SKU)}

O volume semanal do SKU $i$ na semana-alvo $w^*$ é obtido do output do modelo de
otimização:
\begin{equation}\label{eq:dow:volume-modelo}
V^{\text{mod}}_{i,w^*} = x_i^{\text{aloc}} + r_i^{\text{res}},
\quad \forall i \in \mathcal{I}
\end{equation}
onde:
\begin{itemize}
    \item $x_i^{\text{aloc}} \geq 0$: volume alocado pelo otimizador (do excedente
        da classe), coluna \texttt{quantidade} no output do modelo;
    \item $r_i^{\text{res}} \geq 0$: volume reservado por pedido garantido
        ($= 0$ se o SKU não tem pedido), coluna \texttt{quantidade\_reservada}
        no comparador.
\end{itemize}

Há duas situações de reserva:
\begin{enumerate}
    \item \textbf{SKU restrito com pedido} (ex: GRANEL, Exportação): não participa
        da otimização; $x_i^{\text{aloc}}=0$ e $V^{\text{mod}}_{i,w^*}=r_i^{\text{res}}$.
        Aparece como \texttt{tipo=reserva} no output do modelo.
    \item \textbf{SKU permitido com pedido}: participa da otimização do excedente;
        $V^{\text{mod}}_{i,w^*}=x_i^{\text{aloc}}+r_i^{\text{res}}$.
        A reserva é descontada da produção da classe antes da otimização, e o SKU
        pode receber volume adicional visando maximização de margem.
\end{enumerate}

Esses valores são os mesmos exibidos no comparador de produção/alocação
(\texttt{quantidade\_alocada} + \texttt{quantidade\_reservada}), garantindo
rastreabilidade ponta a ponta.

\subsubsection{Histórico de produção}

\begin{equation}\label{eq:dow:historico}
V_{i,c,w,d} \geq 0: \quad \text{volume líquido produzido do SKU } i
    \text{ (classe } c\text{), semana } w\text{, dia } d
\end{equation}

\textbf{Premissa:} o volume histórico deve refletir a \emph{produção líquida} (ACA
menos EAC), obtida via agregação (soma) antes da filtragem de valores positivos.
Os mesmos filtros de estabelecimento e SKUs ativos do pipeline principal devem ser
aplicados ao histórico.

\subsection{Agregações Históricas}

\subsubsection{Agregações por SKU}
\begin{align}\label{eq:dow:agg-sku}
V_{i,\cdot,d} &= \sum_{w \in \mathcal{W}} V_{i,c,w,d},
    && \text{volume do SKU } i \text{ no dia } d \text{ (toda a janela)} \\
V_{i,\cdot,\cdot} &= \sum_{d \in \mathcal{D}} V_{i,\cdot,d},
    && \text{volume total do SKU } i \text{ na janela}
\end{align}

\subsubsection{Agregações por classe (auxiliares para fallback)}
\begin{align}\label{eq:dow:agg-classe}
V_{c,\cdot,d} &= \sum_{i \in \mathcal{I}_c} V_{i,\cdot,d},
    && \text{volume da classe } c \text{ no dia } d \text{ (toda a janela)} \\
V_{c,\cdot,\cdot} &= \sum_{d \in \mathcal{D}} V_{c,\cdot,d},
    && \text{volume total da classe } c \text{ na janela}
\end{align}

\subsection{Perfil Histórico do SKU por Dia da Semana}

Para cada SKU com histórico ($V_{i,\cdot,\cdot}>0$), define-se a participação do
dia $d$ no volume semanal do SKU:
\begin{equation}\label{eq:dow:perfil-sku-dia}
p^{\text{hist}}_{i,d} = \frac{V_{i,\cdot,d}}{V_{i,\cdot,\cdot}},
\quad \forall i : V_{i,\cdot,\cdot}>0,\;\forall d \in \mathcal{D}
\end{equation}

\textbf{Propriedade:} $\displaystyle\sum_{d \in \mathcal{D}} p^{\text{hist}}_{i,d} = 1$,
garantindo que a soma dos 7 dias reproduz exatamente o volume semanal do modelo.

O perfil $p^{\text{hist}}_{i,d}$ responde: ``historicamente, qual percentual da
produção semanal do SKU $i$ cai no dia $d$?''

\subsection{Perfil Histórico da Classe por Dia (Auxiliar para Fallback)}

Para SKUs sem histórico individual, utiliza-se o perfil da classe como fallback:
\begin{equation}\label{eq:dow:perfil-classe-dia}
p^{\text{hist}}_{c,d} = \frac{V_{c,\cdot,d}}{V_{c,\cdot,\cdot}},
\quad \forall c : V_{c,\cdot,\cdot}>0,\;\forall d \in \mathcal{D}
\end{equation}

A Equação~\eqref{eq:dow:perfil-classe-dia} é matematicamente equivalente à média
ponderada por volume semanal:
\[
p^{\text{hist}}_{c,d}
= \sum_{w \in \mathcal{W}^{+}_c}
  \underbrace{\frac{V_{c,w,\cdot}}{\sum_u V_{c,u,\cdot}}}_{\alpha_{c,w}}
  \cdot
  \underbrace{\frac{V_{c,w,d}}{V_{c,w,\cdot}}}_{p_{c,w,d}}
\]
onde $\mathcal{W}^{+}_c := \{w : V_{c,w,\cdot}>0\}$.
Semanas com maior volume têm maior influência no perfil.

\subsection{Distribuição Diária por SKU}

A estimativa de volume diário para cada SKU é:
\begin{equation}\label{eq:dow:distribuicao-diaria}
\hat{V}_{i,w^*,d} = V^{\text{mod}}_{i,w^*} \cdot p^{\text{hist}}_{i,d},
\quad \forall i \in \mathcal{I},\;\forall d \in \mathcal{D}
\end{equation}

onde $p^{\text{hist}}_{i,d}$ é o perfil do SKU
(Equação~\eqref{eq:dow:perfil-sku-dia}) ou, em caso de fallback, o perfil da
classe (Equação~\eqref{eq:dow:perfil-classe-dia}).

\subsection{Condição de Consistência}

\textbf{Fechamento por SKU} (requisito principal):
\begin{equation}\label{eq:dow:consistencia-sku}
\sum_{d \in \mathcal{D}} \hat{V}_{i,w^*,d} = V^{\text{mod}}_{i,w^*},
\quad \forall i \in \mathcal{I}
\end{equation}

A Equação~\eqref{eq:dow:consistencia-sku} garante que a soma dos 7 dias é
\emph{idêntica} ao valor exibido no comparador de produção/alocação.
Isso decorre diretamente de $\sum_d p^{\text{hist}}_{i,d} = 1$.

\textbf{Fechamento por classe} (consequência):
\begin{equation}\label{eq:dow:consistencia-classe}
\sum_{i \in \mathcal{I}_c} \sum_{d \in \mathcal{D}} \hat{V}_{i,w^*,d}
= \sum_{i \in \mathcal{I}_c} V^{\text{mod}}_{i,w^*},
\quad \forall c \in \mathcal{C}
\end{equation}

\subsection{Regras de Fallback}

\subsubsection{F1: SKU sem histórico individual}

Se $V_{i,\cdot,\cdot}=0$ mas a classe $\kappa(i)$ possui histórico
($V_{c,\cdot,\cdot}>0$), utilizar o perfil da classe:
\begin{equation}\label{eq:dow:fallback-f1}
p^{\text{hist}}_{i,d} \leftarrow p^{\text{hist}}_{c,d},
\quad c = \kappa(i),\;\forall d \in \mathcal{D}
\end{equation}
e distribuir:
\begin{equation}\label{eq:dow:distribuicao-fallback-f1}
\hat{V}_{i,w^*,d} = V^{\text{mod}}_{i,w^*} \cdot p^{\text{hist}}_{c,d},
\quad \forall d \in \mathcal{D}
\end{equation}

O perfil menos granular (classe) atua como proxy quando o SKU não possui histórico
individual suficiente. O output deve registrar
\texttt{fallback\_motivo = sku\_sem\_historico\_individual}.

\subsubsection{F2: SKU e classe sem histórico}

Se $V_{i,\cdot,\cdot}=0$ e $V_{c,\cdot,\cdot}=0$ (nem o SKU nem sua classe possuem
histórico na janela), adota-se como premissa de negócio \textbf{não distribuir}:
\begin{equation}\label{eq:dow:fallback-f2}
\hat{V}_{i,w^*,d} \leftarrow 0,
\quad \forall d \in \mathcal{D}
\end{equation}

O volume do modelo é preservado no consolidado semanal, porém a quebra diária é
zerada. O output deve registrar
\texttt{fallback\_motivo = sku\_e\_classe\_sem\_historico}.

\textbf{Nota:} neste cenário, a condição de consistência
(Equação~\eqref{eq:dow:consistencia-sku}) é relaxada; o volume semanal do modelo
não é desagregado por dia, sinalizando ao usuário a necessidade de decisão manual.

\subsection{Observações de Implementação}

\begin{itemize}
    \item \textbf{Origem de $V^{\text{mod}}_{i,w^*}$:} obtido do
        \texttt{resultado.resultado}, combinando \texttt{quantidade}
        (tipo \texttt{otimizacao}) com o valor correspondente em
        \texttt{pedidos\_garantidos\_por\_sku} para SKUs permitidos com pedido.
        Para SKUs restritos (GRANEL, Exportação), o volume é o próprio
        valor das linhas \texttt{tipo=reserva}.
    \item \textbf{Produção líquida:} o histórico deve incorporar ajustes ACA/EAC
        via agregação (soma líquida por item$\times$dia), \emph{antes} de filtrar
        valores não-positivos, evitando superestimação de volumes.
    \item \textbf{Premissa de janela:} parâmetros dedicados
        (\texttt{dow\_ano\_ref}, \texttt{dow\_mes\_ref}, \texttt{dow\_meses\_janela}),
        com fallback para \texttt{ano\_custo}/\texttt{mes\_custo}/\texttt{meses\_janela\_custo}.
    \item A semana alvo $w^*$ deve ser excluída da história para evitar vazamento.
    \item Os mesmos filtros de estabelecimento e SKUs ativos do pipeline principal
        devem ser aplicados.
    \item Execução restrita a granularidade semanal (\texttt{S}).
    \item O output deve conter colunas de rastreabilidade de fallback
        (\texttt{fallback\_aplicado}, \texttt{fallback\_motivo}).
\end{itemize}

    \begin{enumerate}[label=\alph*.]
        \item Carregar produção, classes, pedidos, preços, custos
        \item Carregar e filtrar faturamento (correção de estabelecimento)
        \item Calcular demanda histórica e volume histórico total
        \item Filtrar SKUs válidos (\texttt{skus\_restritos.xlsx})
        \item Processar reservas de pedidos
        \item Calcular produção excedente por classe
    \end{enumerate}
    
    \item \textbf{Otimização} (\texttt{modelo/otimizador.py})
    \begin{enumerate}[label=\alph*.]
        \item Criar solver e configurar parâmetros
        \item Criar variáveis de decisão ($x_i$)
        \item Adicionar restrições (capacidade + demanda)
        \item Resolver e extrair resultados
    \end{enumerate}
    
    \item \textbf{Comparativo Baseline} (\texttt{main.py})
    \begin{enumerate}[label=\alph*.]
        \item Calcular proporção histórica por SKU/classe
        \item Calcular margem baseline
        \item Comparar com margem otimizada
    \end{enumerate}
    
    \item \textbf{Relatório} (\texttt{comparar\_producao\_alocacao.py})
    \begin{enumerate}[label=\alph*.]
        \item Consolidar dados de produção, alocação, reservas
        \item Gerar comparativo detalhado por SKU
    \end{enumerate}
\end{enumerate}

\section{Métricas de Saída}
%==============================================================================

\subsection{Métricas de Volume}

\begin{align}\label{eq:auto:al:010}
\text{Quantidade Alocada Total} &= \sum_{i \in H} x_i^* \\[6pt]
\text{Quantidade Reservada Total} &= \sum_{i \in R} r_i \\[6pt]
\text{Saldo Não Alocado} &= \sum_{c \in C} P_c - \left(\sum_{i \in H} x_i^* + \sum_{i \in R} r_i\right)
\end{align}

\subsection{Métricas Financeiras}

\begin{align}\label{eq:auto:al:011}
\text{Margem Otimizada} &= M^{\text{otim}} = \sum_{i \in H} \mu_i \cdot x_i^* \\[6pt]
\text{Margem Baseline} &= M^{\text{base}} = \sum_{c \in C} \sum_{i} V_c^{\text{otim}} \cdot \pi_i \cdot \mu_i \\[6pt]
\text{Ganho} &= M^{\text{otim}} - M^{\text{base}} \\[6pt]
\text{Receita Total} &= \sum_{i \in H} \frac{p_i}{e_i} \cdot x_i^* \\[6pt]
\text{Custo Total} &= \sum_{i \in H} \frac{c_i}{e_i} \cdot x_i^*
\end{align}
